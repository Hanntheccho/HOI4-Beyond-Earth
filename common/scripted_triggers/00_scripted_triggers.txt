is_minor_tag = {
	OR = {
		original_tag = USA
		original_tag = SIL
		original_tag = LEG
		original_tag = GRP
		original_tag = ONU
		original_tag = KAR
		original_tag = CAL
		original_tag = COS
		original_tag = MLW
		original_tag = GNA
		original_tag = TOG
		original_tag = RWA
		original_tag = BRD
		if = {
			limit = {
				FROM = {
					tag = CRO
				}
			}
			original_tag = YUG
		}
		else_if = {
			limit = {
				FROM = {
					tag = YUG
				}
			}
			original_tag = CRO
		}
		else_if = {
			limit = {
				FROM = {
					is_eng_tag = yes
				}
			}
			is_eng_tag = yes
		}
		else_if = {
			limit = {
				FROM = {
					tag = DOS
				}
			}
			original_tag = CAS
		}
		else_if = {
			limit = {
				FROM = {
					tag = CAS
				}
			}
			original_tag = DOS
		}
		else_if = {
			limit = {
				FROM = {
					tag = CIN
				}
			}
			original_tag = RSA
		}
		else_if = {
			limit = {
				FROM = {
					tag = RSA
				}
			}
			original_tag = CIN
		}
		else = {
			always = no
		}
	}
}

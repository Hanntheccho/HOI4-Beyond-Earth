
calculate_development_level = {
	if = {
		limit = {
			num_of_factories > 119
		}
		add_ideas = global_stage
	}
	else_if = {
		limit = {
			num_of_factories > 79
		}
		add_ideas = continental_stage
	}
	else_if = {
		limit = {
			num_of_factories > 39
		}
		add_ideas = regional_stage
	}
	else_if = {
		limit = {
			num_of_factories > 19
		}
		add_ideas = developed_stage
	}
	else_if = {
		limit = {
			num_of_factories > 4
		}
		add_ideas = developing_stage
	}
	else = {
		add_ideas = agrarian_stage
	}
}

#expectes a temp variable country_to_initiate which should contain original_tag to instantiate as a collabration government
instantiate_collaboration_government = {
	custom_effect_tooltip = compliance_80_effect_tooltip
	hidden_effect = {
		set_variable = { collaboration_formed@var:country_to_initiate = 1 }

		if = {
			# if country_to_initiate does not exist, simply release it as a puppet
			limit = {
				var:country_to_initiate = {
					exists = no
				}
			}
			var:country_to_initiate = { set_variable = { collaboration_formed_by = PREV } }
			release_autonomy = {
				target = country_to_initiate
				autonomy_state = autonomy_puppet
				freedom_level = 0.5
				release_non_owned_controlled = yes
			}
		}
		else = {
			# if country_to_initiate exists, create a dynamic country as our new puppet

			create_dynamic_country = {
				original_tag = country_to_initiate

				set_variable = { collaboration_formed_by = PREV }
				set_temp_variable = { new_country = this }

				PREV = {
					every_controlled_state = {
						limit = { occupied_country_tag = country_to_initiate }

						var:new_country = {
							transfer_state = PREV
						}
					}

					puppet = new_country

					set_autonomy = {
						target = new_country
						autonomy_state = autonomy_puppet
						freedom_level = 0.5
					}
				}
			}
		}
	}
}

		limit = {
			has_dynamic_modifier = {
				modifier = no_ecobalance
			}
		}
		remove_dynamic_modifier = { modifier = no_ecobalance }
		add_dynamic_modifier = { modifier = counter_ecobalance }
	}

		limit = {
			has_dynamic_modifier = {
				modifier = minor_ecobalance
			}
		}

calculate_current_ecobalance = {
	clear_variable = ROOT.current_ecobalance
	every_owned_state = {
		limit = {
			has_dynamic_modifier = {
				modifier = notable_ecobalance
			}
		}
		set_variable = { THIS.state_ecobalance = 1 }
		add_to_variable = { ROOT.current_ecobalance = THIS.state_ecobalance }
	}
	every_owned_state = {
		limit = {
			has_dynamic_modifier = {
				modifier = major_ecobalance
			}
		}
		set_variable = { THIS.state_ecobalance = 2 }
		add_to_variable = { ROOT.current_ecobalance = THIS.state_ecobalance }
	}
	every_owned_state = {
		limit = {
			has_dynamic_modifier = {
				modifier = significant_ecobalance
			}
		}
		set_variable = { THIS.state_ecobalance = 3 }
		add_to_variable = { ROOT.current_ecobalance = THIS.state_ecobalance }
	}
	every_owned_state = {
		limit = {
			has_dynamic_modifier = {
				modifier = terminal_ecobalance
			}
		}
		set_variable = { THIS.state_ecobalance = 4 }

	}
}

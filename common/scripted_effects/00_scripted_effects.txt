build_random_civilian_complex = {
	random_controlled_state = {
		limit = {
			is_fully_controlled_by = ROOT
			free_building_slots = { building = industrial_complex size > 0 include_locked = no }
		}
		add_building_construction = {
			type = industrial_complex
			level = 1 instant_build = yes
		}
	}
}
build_random_military_complex = {
	random_controlled_state = {
		limit = {
			is_fully_controlled_by = ROOT
			free_building_slots = { building = arms_factory size > 0 include_locked = no }
		}
		add_building_construction = { type = arms_factory level = 1 instant_build = yes }
	}
}

calculate_development_level = {
	if = { limit = { num_of_factories > global.num_of_factories_global } add_ideas = global_stage }
	else_if = { limit = { num_of_factories > global.num_of_factories_continental } add_ideas = continental_stage }
	else_if = { limit = { num_of_factories > global.num_of_factories_regional } add_ideas = regional_stage }
	else_if = { limit = { num_of_factories > global.num_of_factories_developed } add_ideas = developed_stage }
	else_if = { limit = { num_of_factories > global.num_of_factories_developing } add_ideas = developing_stage }
	else = { add_ideas = agrarian_stage }
}

puppet_government_too = {
	THIS = {
		if = {
			limit = { is_puppet = yes }
			load_focus_tree = generic_focus
			add_ideas = volunteer_only

			if = { limit = { THIS.OVERLORD = { has_idea = oligarchic_government } } add_ideas = oligarchic_government }
			else_if = { limit = { THIS.OVERLORD = { has_idea = minarchist_state } } add_ideas = minarchist_state }
			else_if = { limit = { THIS.OVERLORD = { has_idea = digital_democracy } } add_ideas = digital_democracy }
			else_if = { limit = { THIS.OVERLORD = { has_idea = communal_parity } } add_ideas = communal_parity }
			else_if = { limit = { THIS.OVERLORD = { has_idea = autocratic_authority } } add_ideas = autocratic_authority }

			if = { limit = { THIS.OVERLORD = { has_idea = isolation } } add_ideas = isolation }
			else_if = { limit = { THIS.OVERLORD = { has_idea = civilian_economy } } add_ideas = civilian_economy }
			else_if = { limit = { THIS.OVERLORD = { has_idea = low_economic_mobilisation } } add_ideas = low_economic_mobilisation }
			else_if = { limit = { THIS.OVERLORD = { has_idea = partial_economic_mobilisation } } add_ideas = partial_economic_mobilisation }
			else_if = { limit = { THIS.OVERLORD = { has_idea = war_economy } } add_ideas = war_economy }
			else_if = { limit = { THIS.OVERLORD = { has_idea = tot_economic_mobilisation } } add_ideas = tot_economic_mobilisation }

			if = { limit = { THIS.OVERLORD = { has_idea = free_trade } } add_ideas = free_trade }
			else_if = { limit = { THIS.OVERLORD = { has_idea = export_focus } } add_ideas = export_focus }
			else_if = { limit = { THIS.OVERLORD = { has_idea = limited_exports } } add_ideas = limited_exports }
			else_if = { limit = { THIS.OVERLORD = { has_idea = closed_economy } } add_ideas = closed_economy }
		}
	}
}

calculate_ecobalance_status = {
	if = { limit = { check_variable = { ecobalance_ratio < 0.01 } } add_ideas = optimum_ecobalance }
	else_if = { limit = { check_variable = { ecobalance_ratio < 0.2 } } add_ideas = favorable_ecobalance }
	else_if = { limit = { check_variable = { ecobalance_ratio < 0.4 } } add_ideas = neutral_ecobalance }
	else_if = { limit = { check_variable = { ecobalance_ratio < 0.6 } } add_ideas = precarious_ecobalance }
	else_if = { limit = { check_variable = { ecobalance_ratio < 0.8 } } add_ideas = dangerous_ecobalance }
	else = { add_ideas = disastrous_ecobalance }
}

increase_ecobalance = {
	if = { limit = { has_dynamic_modifier = { modifier = no_ecobalance } }
		remove_dynamic_modifier = { modifier = no_ecobalance }
		add_dynamic_modifier = { modifier = minor_ecobalance }
	}
	else_if = { limit = { has_dynamic_modifier = { modifier = minor_ecobalance } }
		remove_dynamic_modifier = { modifier = minor_ecobalance }
		add_dynamic_modifier = { modifier = notable_ecobalance }
	}
	else_if = { limit = { has_dynamic_modifier = { modifier = notable_ecobalance } }
		remove_dynamic_modifier = { modifier = notable_ecobalance }
		add_dynamic_modifier = { modifier = major_ecobalance }
	}
	else_if = { limit = { has_dynamic_modifier = { modifier = major_ecobalance } }
		remove_dynamic_modifier = { modifier = major_ecobalance }
		add_dynamic_modifier = { modifier = significant_ecobalance }
	}
	else_if = { limit = { has_dynamic_modifier = { modifier = significant_ecobalance } }
		remove_dynamic_modifier = { modifier = significant_ecobalance }
		add_dynamic_modifier = { modifier = terminal_ecobalance }
	}
}

decrease_ecobalance = {
	if = {
		limit = {
			ROOT = { has_government = mastery }
			OR = {
				has_dynamic_modifier = { modifier = no_ecobalance }
				has_dynamic_modifier = { modifier = minor_ecobalance }
				has_dynamic_modifier = { modifier = notable_ecobalance }
				has_dynamic_modifier = { modifier = major_ecobalance }
				has_dynamic_modifier = { modifier = significant_ecobalance }
				has_dynamic_modifier = { modifier = terminal_ecobalance }
			}
		}
		hidden_effect = {
			remove_dynamic_modifier = { modifier = no_ecobalance }
			remove_dynamic_modifier = { modifier = minor_ecobalance }
			remove_dynamic_modifier = { modifier = notable_ecobalance }
			remove_dynamic_modifier = { modifier = major_ecobalance }
			remove_dynamic_modifier = { modifier = significant_ecobalance }
			remove_dynamic_modifier = { modifier = terminal_ecobalance }
		}
		add_dynamic_modifier = { modifier = counter_ecobalance }
	}
	else_if = { limit = { ROOT = { has_government = harmony } has_dynamic_modifier = { modifier = counter_ecobalance } }
		remove_dynamic_modifier = { modifier = counter_ecobalance }
		add_dynamic_modifier = { modifier = no_ecobalance }
	}
	else_if = { limit = { ROOT = { has_government = harmony } has_dynamic_modifier = { modifier = minor_ecobalance } }
		remove_dynamic_modifier = { modifier = minor_ecobalance }
		add_dynamic_modifier = { modifier = no_ecobalance }
	}
	else_if = { limit = { has_dynamic_modifier = { modifier = notable_ecobalance } }
		remove_dynamic_modifier = { modifier = notable_ecobalance }
		add_dynamic_modifier = { modifier = minor_ecobalance }
	}
	else_if = { limit = { has_dynamic_modifier = { modifier = major_ecobalance } }
		remove_dynamic_modifier = { modifier = major_ecobalance }
		add_dynamic_modifier = { modifier = notable_ecobalance }
	}
	else_if = { limit = { has_dynamic_modifier = { modifier = significant_ecobalance } }
		remove_dynamic_modifier = { modifier = significant_ecobalance }
		add_dynamic_modifier = { modifier = major_ecobalance }
	}
	else_if = { limit = { has_dynamic_modifier = { modifier = terminal_ecobalance } }
		remove_dynamic_modifier = { modifier = terminal_ecobalance }
		add_dynamic_modifier = { modifier = significant_ecobalance }
	}
}

calculate_current_ecobalance = {
	clear_variable = ROOT.current_ecobalance
	every_owned_state = {
		if = { limit = { has_dynamic_modifier = { modifier = notable_ecobalance } }
			add_to_variable = { ROOT.current_ecobalance = 1 }
		}
		else_if = { limit = { has_dynamic_modifier = { modifier = major_ecobalance } }
			add_to_variable = { ROOT.current_ecobalance = 2 }
		}
		else_if = { limit = { has_dynamic_modifier = { modifier = significant_ecobalance } }
			add_to_variable = { ROOT.current_ecobalance = 3}
		}
		else_if = { limit = { has_dynamic_modifier = { modifier = terminal_ecobalance } }
			add_to_variable = { ROOT.current_ecobalance = 4 }
		}
	}
}

###############################
########## MANPOWER ###########
###############################

on_actions = {
	on_startup = {
		effect = {
			every_possible_country = {
				limit = {
					is_dynamic_country = no
				}
				if = {
					limit = {
						tag = ADU
					}
					add_ideas = volunteer_only
				}
				else_if = {
					limit = {
						exists = yes #all non-existing countries have 0 manpower
						has_manpower < 1000
					}
					add_ideas = service_by_requirement
				}
				else_if = {
					limit = {
						OR = {
							has_idea = communal_parity
							has_idea = autocratic_authority
						}
					}
					add_ideas = extensive_conscription
				}
				else_if = {
					limit = {
						OR = {
							has_idea = digital_democracy
							has_idea = minarchist_state
						}
					}
					add_ideas = volunteer_only
				}
				else = { add_ideas = limited_conscription }
			}
		}
	}
}

###### INDUSTRIAL POWER #######

on_actions = {
	on_daily = {
		effect = {
			set_variable = { industrial_power_gain = 0 }
			set_variable = { industrial_power_civilian_factories = num_of_civilian_factories }
			divide_variable = { industrial_power_civilian_factories = 5 }
			set_variable = { industrial_power_military_factories = num_of_military_factories }
			divide_variable = { industrial_power_military_factories = 10 }
			multiply_variable = { industrial_power_military_factories = -1 }
			set_variable = { industrial_power_spaceship_factories = 0 }
			every_controlled_state = {
				add_to_variable = {
					PREV.industrial_power_spaceship_factories = building_level@commercial_nuclear_reactor
				}
			}
			divide_variable = { industrial_power_spaceship_factories = 5 }
			multiply_variable = { industrial_power_spaceship_factories = -1 }
			add_to_variable = { industrial_power_gain = 1 }
			add_to_variable = { industrial_power_gain = industrial_power_civilian_factories }
			add_to_variable = { industrial_power_gain = industrial_power_military_factories }
			add_to_variable = { industrial_power_gain = industrial_power_spaceship_factories }
			add_to_variable = { industrial_power_gain = modifier@industrial_power_gain }
			set_variable = { industrial_power_rate = modifier@industrial_power_rate }
			add_to_variable = { industrial_power_rate = 1 }
			clamp_variable = { var = industrial_power_rate min = 0 }
			multiply_variable = { industrial_power_gain = industrial_power_rate }
			add_to_variable = { industrial_power = industrial_power_gain }
			clamp_variable = { var = industrial_power min = -999 max = 1000 }
		}
	}
}

###############################
######### FREE FACTORY ########
###############################

on_actions = {
	on_daily = {
		effect = {
			set_variable = { free_factory_adjustment = modifier@free_factory_adjustment }
			if = {
				limit = { country_is_missing_any_resource = yes }
				multiply_variable = { free_factory_adjustment = 0.1 }
			}
			set_variable = { free_factory_gain = free_factory_base_gain }
			clamp_variable = { var = free_factory_adjustment min = -1 }
			add_to_variable = { free_factory_adjustment = 1 }
			multiply_variable = { free_factory_gain = free_factory_adjustment }
			add_to_variable = { free_factory_progress = free_factory_gain }
			if = {
				limit = { check_variable = { free_factory_progress > 100 } }
				add_to_variable = { free_factory_counter = 1 }
				random_list = {
					50 = {
						build_random_civilian_complex = yes
					}
					50 = {
						build_random_military_complex = yes
					}
				}
				clear_variable = free_factory_progress
				set_variable = { free_factory_progress = 0 }
			}
		}
	}
}

					limit = { owns_state = 333 }
					add_to_variable = { global.threat_debug = global.total_global_ecobalance }
				}
			}
		}
	}

				}

						limit = { PREV = { has_government = mastery } NOT = { has_dynamic_modifier = { modifier = counter_ecobalance } } }

						set_state_flag = { flag = ecobalance_reduction_timeout_flag value = 1 days = 120 }

						limit = { PREV = { has_government = harmony } NOT = { has_dynamic_modifier = { modifier = no_ecobalance } } }

						set_state_flag = { flag = ecobalance_reduction_timeout_flag value = 1 days = 30 }

						limit = {
							NOT = {
								has_dynamic_modifier = { modifier = counter_ecobalance }

						set_state_flag = { flag = ecobalance_reduction_timeout_flag value = 1 days = 60 }
						decrease_ecobalance = yes
					}
				}
			}
		}
	}
}
